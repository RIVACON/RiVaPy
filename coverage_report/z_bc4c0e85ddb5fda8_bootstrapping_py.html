459def find_bracket(error_fn, guess, args, expand=2.0, max_tries=10, min_lower=1e-8, max_upper=10.0): 

460 """ 

461 Tries to find a [lower, upper] bracket where error_fn(lower) and error_fn(upper) 

462 have opposite signs, indicating a root lies between them. 

463 

464 Parameters 

465 ---------- 

466 error_fn : callable 

467 Your pricing error function (same as passed to brentq). 

468 guess : float 

469 A rough estimate for the root (e.g., previous DF). 

470 args : tuple 

471 Extra arguments passed to error_fn. 

472 expand : float 

473 Factor by which to widen the bracket each iteration. 

474 max_tries : int 

475 Maximum number of expansions before giving up. 

476 min_lower, max_upper : float 

477 Hard limits to keep brackets within safe numeric bounds. 

478 """ 

479 lower = max(guess * 0.8, min_lower) 

480 upper = min(guess * 1.2, max_upper) 

481 

482 f_lower = error_fn(lower, *args) 

483 f_upper = error_fn(upper, *args) 

484 

485 tries = 0 

486 while f_lower * f_upper > 0 and tries < max_tries: 

487 # Expand symmetrically outward 

488 lower = max(lower / expand, min_lower) 

489 upper = min(upper * expand, max_upper) 

490 f_lower = error_fn(lower, *args) 

491 f_upper = error_fn(upper, *args) 

492 tries += 1 

493 

494 if f_lower * f_upper > 0: 

495 raise RuntimeError("Could not find valid bracket for brentq") 

496 

497 return lower, upper
